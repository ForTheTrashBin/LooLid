func localizeStatError(err error, path string) string {
	switch {
	case os.IsNotExist(err):
		return t("file_not_found", path)

	case os.IsPermission(err):
		return t("permission_denied", path)

	case errors.Is(err, syscallENOTDIR()):
		return t("not_directory", path)

	default:
		return t("stat_failed", path)
	}
}

Dann ist die nutzerfreundliche Meldung übersetzt, während der Originalfehler für Debugging erhalten bleibt.
Kannst Du das Beispiel bitte auf ALLE System-Fehlermeldungen erweitern!

Nicht sinnvoll als „alle Texte übersetzen“. In Go gibt es keine stabile, vollständige Liste aller möglichen err.Error()-Systemmeldungen: sie kommen vom Betriebssystem, sind je nach OS/Version/Locale unterschiedlich und sollten nicht per Text gemappt werden.

Der richtige Weg ist: Errno/Fehlerklasse erkennen und dafür eigene lokalisierte Meldungen ausgeben. Für sehr viele Systemfehler kannst du syscall.Errno behandeln.

Beispiel für Unix/Linux/macOS-artige Systeme:

func localizeError(err error) string {
	if err == nil {
		return ""
	}

	var errno syscall.Errno
	if errors.As(err, &errno) {
		if msg, ok := deErrno[errno]; ok {
			return msg
		}
		return fmt.Sprintf("Systemfehler %d", errno)
	}

	if os.IsNotExist(err) {
		return "Datei oder Verzeichnis nicht gefunden"
	}
	if os.IsPermission(err) {
		return "Keine Berechtigung"
	}

	return "Ein unbekannter Fehler ist aufgetreten"
}

func main() {
	path := "/root/test.txt"

	_, err := os.Stat(path)
	if err != nil {
		fmt.Printf("%s: %s\n", localizeError(err), path)

		// Optional für Debugging:
		fmt.Printf("Technische Details: %v\n", err)
		os.Exit(1)
	}
}

var errno syscall.Errno

func runStep(name string, duration time.Duration, sigCh chan os.Signal) error {
	cfg := yacspin.Config{
		Frequency:         100 * time.Millisecond,
		CharSet:           yacspin.CharSets[14],
		Suffix:            " " + name,
		SuffixAutoColon:   true,
		StopCharacter:     "✓",
		StopColors:        []string{"fgGreen"},
		StopFailCharacter: "x",               // TODO: Mist
		StopFailColors:    []string{"fgRed"}, // TODO: Mist
		StopMessage:       "done",
	}

	spinner, err := yacspin.New(cfg)
	if err != nil {
		return fmt.Errorf("spinner init failed: %w", err)
	}

	if err := spinner.Start(); err != nil {
		return fmt.Errorf("spinner start failed: %w", err)
	}

	defer spinner.Stop()

	doneCh := make(chan error, 1)

	// simulierte Arbeit + Fehlerchance
	go func() {
		timer := time.NewTimer(duration)
		defer timer.Stop()

		<-timer.C

		// Beispiel: künstlicher Fehler in Step 2
		if name == "Step 4: proccessing acknowledge" {
			doneCh <- errors.New("simulated processing error")
			return
		}

		doneCh <- nil
	}()

	select {
	case <-sigCh:
		spinner.StopFailMessage("aborted by user")
		spinner.StopFail()
		return ErrInterrupted

	case err := <-doneCh:
		if err != nil {
			spinner.StopFailMessage(err.Error())
			spinner.StopFail()
			return err
		}
		return nil
	}
}

var ErrInterrupted = errors.New("interrupted by user")

func main() {
	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, os.Interrupt, syscall.SIGTERM)
	defer signal.Stop(sigCh)

	if err := runStep("Step 1: loading data", 3*time.Second, sigCh); err != nil {
		handleError("Step 1", err)
		return
	}

	if err := runStep("Step 2: processing data", 3*time.Second, sigCh); err != nil {
		handleError("Step 2", err)
		return
	}

	if err := runStep("Step 3: sending data", 3*time.Second, sigCh); err != nil {
		handleError("Step 3", err)
		return
	}

	if err := runStep("Step 4: proccessing acknowledge", 3*time.Second, sigCh); err != nil {
		handleError("Step 5", err)
		return
	}

	if err := runStep("Step 5: saving results", 3*time.Second, sigCh); err != nil {
		handleError("Step 5", err)
		return
	}
}
